-- 1. Create database
CREATE DATABASE healthcare;
USE healthcare;

-- 2. Create tables
CREATE TABLE department (
  departmentid INT PRIMARY KEY,
  departmentname VARCHAR(100) NOT NULL,
  location VARCHAR(100)
);

CREATE TABLE insurance (
  insuranceid INT PRIMARY KEY,
  providername VARCHAR(100),
  policynumber VARCHAR(100),
  approvalstatus VARCHAR(50)
);

CREATE TABLE patient (
  patientid INT PRIMARY KEY,
  insuranceid INT,
  fullname VARCHAR(150) NOT NULL,
  dob DATE,
  age INT,
  gender VARCHAR(10),
  bloodgroup VARCHAR(5),
  email VARCHAR(100),
  address VARCHAR(100),
  FOREIGN KEY (insuranceid) REFERENCES insurance(insuranceid)
);

CREATE TABLE doctor (
  doctorid INT PRIMARY KEY,
  departmentid INT,
  fullname VARCHAR(150),
  gender VARCHAR(10),
  mobileno VARCHAR(20),
  specialization VARCHAR(100),
  experienceyears INT,
  FOREIGN KEY (departmentid) REFERENCES department(departmentid)
);

CREATE TABLE appointment (
  appointmentid INT PRIMARY KEY,
  patientid INT,
  doctorid INT,
  appointmentdatetime DATETIME,
  reason VARCHAR(100),
  FOREIGN KEY (patientid) REFERENCES patient(patientid),
  FOREIGN KEY (doctorid) REFERENCES doctor(doctorid)
);

CREATE TABLE encounter (
  encounterid INT PRIMARY KEY,
  appointmentid INT,
  encounterdatetime DATETIME,
  actual_reason VARCHAR(100),
  patientid INT,
  doctorid INT,
  FOREIGN KEY (appointmentid) REFERENCES appointment(appointmentid),
  FOREIGN KEY (patientid) REFERENCES patient(patientid),
  FOREIGN KEY (doctorid) REFERENCES doctor(doctorid)
);

CREATE TABLE prescription (
  prescriptionid INT PRIMARY KEY,
  encounterid INT,
  prescriptiondate DATE,
  FOREIGN KEY (encounterid) REFERENCES encounter(encounterid)
);

CREATE TABLE medication (
  medicationid INT PRIMARY KEY,
  medicationname VARCHAR(100),
  manufacturer VARCHAR(100)
);

CREATE TABLE prescriptionmedication (
  prescriptionmedicationid INT PRIMARY KEY,
  prescriptionid INT,
  medicationid INT,
  dosage VARCHAR(50),
  duration VARCHAR(50),
  FOREIGN KEY (prescriptionid) REFERENCES prescription(prescriptionid),
  FOREIGN KEY (medicationid) REFERENCES medication(medicationid)
);

CREATE TABLE testtype (
  testtypeid INT PRIMARY KEY,
  testtypename VARCHAR(100)
);

CREATE TABLE diagnosis (
  diagnosisid INT PRIMARY KEY,
  encounterid INT,
  diagnosisdate DATE,
  diagnosisdescription VARCHAR(100),
  FOREIGN KEY (encounterid) REFERENCES encounter(encounterid)
);

CREATE TABLE disease (
  diseaseid INT PRIMARY KEY,
  diseasename VARCHAR(100),
  description VARCHAR(100)
);

CREATE TABLE diagnosisdisease (
  diagdiseaseid INT PRIMARY KEY,
  diagnosisid INT,
  diseaseid INT,
  FOREIGN KEY (diagnosisid) REFERENCES diagnosis(diagnosisid),
  FOREIGN KEY (diseaseid) REFERENCES disease(diseaseid)
);

CREATE TABLE proceduremaster (
  procedureid INT PRIMARY KEY,
  procedurename VARCHAR(100),
  proceduretype VARCHAR(100),
  procedurecost DECIMAL(10,2),
  description VARCHAR(100)
);

CREATE TABLE encounterprocedure (
  encprocid INT PRIMARY KEY,
  encounterid INT,
  procedureid INT,
  proceduredate DATETIME,
  resultnotes VARCHAR(50),
  FOREIGN KEY (encounterid) REFERENCES encounter(encounterid),
  FOREIGN KEY (procedureid) REFERENCES proceduremaster(procedureid)
);

CREATE TABLE billing (
  invoiceid INT PRIMARY KEY,
  encounterid INT,
  invoicedate DATE,
  totalamount DECIMAL(10,2),
  paidamount DECIMAL(10,2),
  paymentstatus VARCHAR(50),
  FOREIGN KEY (encounterid) REFERENCES encounter(encounterid)
);
